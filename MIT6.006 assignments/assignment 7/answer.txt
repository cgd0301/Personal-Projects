(t) STOCKLIMITED(total, count, start, end, limit) {
    create table purchaseAmount
    create table profit
    for cash from 0 to total: {
        profit[cash, 0] = total
        purchaseAmount[cash, 0] = 0
        for stock from 1 to count {
            profit[cash, stock] = max{profit[cash - k * start[stock], stock - 1] + k * (end[stock] - start[stock])
                                    for k from 0 to limit[stock] where cash - k * start[stock] >= 0}
            purchaseAmount[cash, stock] = k which leads to the largest profit in the recurrence function
        }
    }
    return profit[total, count]
}
